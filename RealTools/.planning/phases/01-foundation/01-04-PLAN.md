**File 2 — `components/sidebar-nav.tsx`** (Client Component for active state via usePathname):
```typescript
'use client'

import Link from 'next/link'
import { usePathname } from 'next/navigation'
import { Briefcase, Users, User } from 'lucide-react'
import { cn } from '@/lib/utils'

const navItems = [
  { href: '/dashboard', label: 'Deals', icon: Briefcase, matchPrefix: '/dashboard' },
  { href: '/buyers', label: 'Buyers', icon: Users, matchPrefix: '/buyers' },
  { href: '/profile', label: 'Profile', icon: User, matchPrefix: '/profile' },
]

export function SidebarNav() {
  const pathname = usePathname()

  return (
    <nav className="flex-1 px-2 py-4 space-y-1">
      {navItems.map(({ href, label, icon: Icon, matchPrefix }) => {
        const isActive = pathname === matchPrefix || pathname.startsWith(`${matchPrefix}/`)
        return (
          <Link
            key={href}
            href={href}
            className={cn(
              'flex items-center gap-2 px-4 py-3 rounded-md text-base min-h-[44px] transition-colors',
              isActive
                ? 'bg-zinc-800 text-zinc-50'
                : 'text-zinc-400 hover:text-zinc-50 hover:bg-zinc-800'
            )}
          >
            <Icon className="size-4" />
            <span>{label}</span>
          </Link>
        )
      })}
    </nav>
  )
}
```

(Lucide icons: `Briefcase` for Deals, `Users` for Buyers, `User` for Profile — matches UI-SPEC line 127's "Lucide React icons at 16px (size-4)". Icon choice is Claude's discretion within the lucide-react bundled with shadcn — these three are unambiguous matches per UI semantic.)

```typescript
'use client'

import { LogOut } from 'lucide-react'
import { useRouter } from 'next/navigation'
import { createSupabaseBrowserClient } from '@/lib/supabase/client'
import { Button } from '@/components/ui/button'

export function LogoutButton() {
  const router = useRouter()

  async function handleLogout() {
    const supabase = createSupabaseBrowserClient()
    await supabase.auth.signOut()
    router.push('/auth/login')
    router.refresh()
  }

  return (
    <Button
      variant="ghost"
      onClick={handleLogout}
      className="w-full justify-start gap-2 text-sm text-zinc-400 hover:text-zinc-50 hover:bg-zinc-800"
    >
      <LogOut className="size-4" />
      Log out
    </Button>
  )
}
```

(Added `router.refresh()` after `push` so the layout's getUser() guard re-evaluates with the cleared cookies — without it, the layout could briefly render stale user data before the next navigation.)

```typescript
import { redirect } from 'next/navigation'
import { createSupabaseServerClient } from '@/lib/supabase/server'
import { Sidebar } from '@/components/sidebar'

export default async function AppLayout({
  children,
}: {
  children: React.ReactNode
}) {
  const supabase = await createSupabaseServerClient()
  // ALWAYS getUser() — defense in depth even though middleware also checks (CLAUDE.md rule).
  const {
    data: { user },
  } = await supabase.auth.getUser()

  if (!user) {
    redirect('/auth/login')
  }

  return (
    <div className="flex h-screen overflow-hidden bg-zinc-950 text-zinc-50">
      <Sidebar />
      <main className="flex-1 overflow-y-auto bg-zinc-900 p-8">
        {children}
      </main>
    </div>
  )
}
```

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| Browser cookie -> Server Component (layout) | Defense-in-depth: middleware AND layout both validate user |
| Browser sign-out -> Supabase Auth | Cookie clearing happens via supabase.auth.signOut(); client must trust the result |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-04-01 | S (Spoofing) | app/(app)/layout.tsx | mitigate | Layout calls `supabase.auth.getUser()` (server-side JWT validation) — even if middleware is misconfigured or a route somehow bypasses it, the layout guard catches it. CLAUDE.md defense-in-depth rule. Acceptance criteria asserts the call exists and `getSession` does NOT. |
| T-04-02 | I (Information disclosure) | app/(app)/profile/page.tsx | mitigate | Profile page reads `user.email` from `getUser()` — the user always sees their OWN email (no cross-user leak possible because RLS isn't even involved; auth.users is queried via the user's own JWT). |
| T-04-03 | T (Tampering) | LogoutButton | mitigate | Sign-out is a Supabase SDK call from the browser using the anon key — the SDK clears cookies and invalidates the session server-side. `router.refresh()` ensures the layout re-evaluates with cleared cookies. |
| T-04-04 | E (Elevation of privilege) | Stale session after logout | mitigate | `router.push('/auth/login')` followed by `router.refresh()` invalidates the React cache for the (app) route group; the next render runs the layout's getUser() guard against the cleared cookie. |
| T-04-05 | R (Repudiation) | Logout event | accept | No audit log of logout events in v1. Sign-in events are auto-logged by Supabase Auth; sign-outs are not. Sufficient for MVP. |
| T-04-06 | D (Denial of service) | Sidebar render | accept | Pure UI render with no external calls. Negligible risk. |
</threat_model>
